function verifierDansGoogleSheet(code) { // Nettoyage intelligent : si la caméra lit une URL complète, on extrait l'ID try { if (code.includes('?')) { const urlObj = new URL(code); code = urlObj.searchParams.get('id') || urlObj.searchParams.get('code') || code; } } catch (e) { // Si ce n'est pas une URL valide, on garde le texte brut scanné } // TRÈS IMPORTANT : Nettoyage des espaces invisibles ou retours à la ligne potentiels code = code.trim(); const container = document.getElementById('result-container'); const statusEl = document.getElementById('result-status'); const nameEl = document.getElementById('result-name'); const codeEl = document.getElementById('result-code'); const btn = document.getElementById('restart-btn'); container.style.display = 'block'; statusEl.innerText = "Vérification..."; nameEl.innerText = ""; codeEl.innerText = "Code : " + code; container.className = ""; // Ajout du paramètre callback=? pour éviter les blocages CORS stricts de Google Apps Script fetch(`${WEB_APP_URL}?code=${encodeURIComponent(code)}&action=entree`, { method: 'GET', mode: 'cors' }) .then(response => response.json()) .then(data => { if (data.status === "ok") { container.className = "bg-success"; statusEl.innerText = "✔ ACCÈS AUTORISÉ"; nameEl.innerText = data.name; } else if (data.status === "already_entered") { container.className = "bg-already"; statusEl.innerText = "⚠ DÉJÀ POINTÉ"; nameEl.innerText = `${data.name} est déjà entré.`; } else { container.className = "bg-unknown"; statusEl.innerText = "✖ INCONNU"; nameEl.innerText = "Ce QR code n'existe pas."; } codeEl.innerText = "ID : " + code; btn.style.display = 'inline-block'; }) .catch(err => { container.className = "bg-unknown"; statusEl.innerText = "Erreur de communication"; nameEl.innerText = "Vérifiez votre connexion ou l'URL du script."; codeEl.innerText = "ID : " + code; btn.style.display = 'inline-block'; console.error(err); }); }
